pipeline {
  agent { label 'myll_ia_bakastov' }

  options {
    timestamps()
    ansiColor('xterm')
    timeout(time: 10, unit: 'MINUTES')
  }

  environment {
    VENV = ".venv"
  }

  stages {
    stage('Checkout') {
      steps {
        checkout scm
      }
    }

    stage('Prepare venv + deps') {
      steps {
        sh '''
          set -e
          python3 -V
          python3 -m venv "$VENV"
          . "$VENV/bin/activate"
          pip install -U pip
          pip install -r requirements.txt
        '''
      }
    }

    stage('Clean ports (8092-8095)') {
      steps {
        sh '''
          set +e
          for p in 8092 8093 8094 8095; do
            fuser -k ${p}/tcp 2>/dev/null || true
          done
          set -e
        '''
      }
    }

    stage('Run smoke test') {
      steps {
        sh '''
          set -e
          mkdir -p run_output

          . "$VENV/bin/activate"

          python Simulator/simulator.py  > run_output/simulator.log 2>&1 &
          SIM_PID=$!
          sleep 3

          python Reciever/reciever.py    > run_output/reciever.log  2>&1 &
          REC_PID=$!
          sleep 3

          python Business/business.py    > run_output/business.log  2>&1 &
          BUS_PID=$!

          sleep 30

          kill $BUS_PID $REC_PID $SIM_PID 2>/dev/null || true
          sleep 2

          tar -czf artifacts.tgz run_output Reciever/*.csv Business/*.csv 2>/dev/null || tar -czf artifacts.tgz run_output
        '''
      }
    }
  }

  post {
    always {
      sh '''
        set +e
        for p in 8092 8093 8094 8095; do
          fuser -k ${p}/tcp 2>/dev/null || true
        done
        set -e
      '''
      archiveArtifacts artifacts: 'artifacts.tgz, run_output/*.log, Reciever/*.csv, Business/*.csv', fingerprint: true
    }
  }
}
